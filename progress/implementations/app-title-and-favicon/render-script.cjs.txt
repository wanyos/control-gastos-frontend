const { chromium } = require('C:/Users/roybe/Escritorio/proyectos/control-gastos/gastos-frontend/node_modules/@playwright/test')
const fs = require('fs')
const root = 'C:/Users/roybe/Escritorio/proyectos/control-gastos/gastos-frontend/'
const out = root + 'progress/implementation/app-title-and-favicon/'
const b64 = (p) => 'data:image/svg+xml;base64,' + Buffer.from(fs.readFileSync(root + p, 'utf8')).toString('base64')
const variants = { green: b64('design-system/assets/logo-mark.svg'), dark: b64('design-system/assets/logo-mark-dark.svg'), chosen: b64('public/favicon.svg') }
// Tab-strip / active-tab colors of Chrome (light and dark themes)
const bgs = [['light strip', '#DEE1E6'], ['light tab', '#FFFFFF'], ['dark strip', '#202124'], ['dark tab', '#35363A']]
const lum = (hex) => { const c = [1,3,5].map(i => parseInt(hex.slice(i,i+2),16)/255).map(v => v <= 0.03928 ? v/12.92 : ((v+0.055)/1.055)**2.4); return 0.2126*c[0]+0.7152*c[1]+0.0722*c[2] }
const ratio = (a,b) => { const [x,y] = [lum(a),lum(b)].sort((m,n)=>n-m); return ((x+0.05)/(y+0.05)).toFixed(2) }
;(async () => {
  const browser = await chromium.launch()
  // 1) Evidence: real-size (DPR 1) rendering at 16 and 32 px on light and dark tab bars
  const page = await browser.newPage({ deviceScaleFactor: 1, viewport: { width: 760, height: 420 } })
  const rows = bgs.map(([name, bg]) => `<div class="row" style="background:${bg};color:${lum(bg) > 0.3 ? '#202124' : '#E8EAED'}">
    <span class="lbl">${name} ${bg}</span>
    ${['green','dark'].map(v => `<span class="cell"><img src="${variants[v]}" width="16" height="16"><img src="${variants[v]}" width="32" height="32"><span class="tab">Control Accounts</span><em>${v}</em></span>`).join('')}
  </div>`).join('')
  await page.setContent(`<style>body{margin:0;font:12px system-ui} .row{display:flex;align-items:center;gap:24px;padding:10px 12px;height:60px} .lbl{width:130px} .cell{display:flex;align-items:center;gap:8px;width:280px} em{opacity:.7}</style>${rows}`)
  await page.waitForTimeout(300)
  await page.screenshot({ path: out + 'tabs-1x.png', fullPage: true })
  // 2) Zoomed (nearest-neighbour x8) view of the 16 px raster, to judge legibility pixel by pixel
  const zoom = await browser.newPage({ deviceScaleFactor: 1, viewport: { width: 900, height: 700 } })
  await zoom.setContent(`<body style="margin:0;font:12px system-ui"><div id="w"></div><script>
    const bgs=${JSON.stringify(bgs)}; const v=${JSON.stringify({green:variants.green,dark:variants.dark})};
    const w=document.getElementById('w');
    const load=s=>new Promise(r=>{const i=new Image();i.onload=()=>r(i);i.src=s});
    (async()=>{for(const [n,bg] of bgs){const row=document.createElement('div');row.style.cssText='display:flex;gap:24px;align-items:center;padding:10px;background:'+bg;
      const l=document.createElement('span');l.textContent=n;l.style.cssText='width:110px;color:'+(n.startsWith('light')?'#202124':'#E8EAED');row.append(l);
      for(const k of ['green','dark']){const img=await load(v[k]);const s=document.createElement('canvas');s.width=s.height=16;const sc=s.getContext('2d');sc.fillStyle=bg;sc.fillRect(0,0,16,16);sc.drawImage(img,0,0,16,16);
        const c=document.createElement('canvas');c.width=c.height=128;const cc=c.getContext('2d');cc.imageSmoothingEnabled=false;cc.drawImage(s,0,0,128,128);row.append(c)}
      w.append(row)} document.title='ready'})()</script></body>`)
  await zoom.waitForFunction(() => document.title === 'ready')
  await zoom.screenshot({ path: out + 'zoom-16px-x8.png', fullPage: true })
  // 3) Rasterize the chosen SVG: 32 px PNG fallback and 180 px apple-touch-icon (opaque, full-bleed: iOS applies its own mask)
  const r = await browser.newPage({ deviceScaleFactor: 1, viewport: { width: 200, height: 200 } })
  await r.setContent(`<style>html,body{margin:0;background:transparent}</style><img id="i" src="${variants.chosen}" width="32" height="32" style="display:block">`)
  await r.waitForTimeout(200)
  await r.locator('#i').screenshot({ path: root + 'public/favicon-32x32.png', omitBackground: true })
  const touch = fs.readFileSync(root + 'public/favicon.svg', 'utf8').replace(' rx="16"', '')
  await r.setContent(`<style>html,body{margin:0}</style><img id="i" src="data:image/svg+xml;base64,${Buffer.from(touch).toString('base64')}" width="180" height="180" style="display:block">`)
  await r.waitForTimeout(200)
  await r.locator('#i').screenshot({ path: root + 'public/apple-touch-icon.png' })
  await browser.close()
  const report = bgs.map(([n,bg]) => `${n} ${bg}: green #0A8F5F -> ${ratio('#0A8F5F',bg)}:1 | dark #0B1116 -> ${ratio('#0B1116',bg)}:1`).join('\n')
  console.log(report + `\nglyph white on green: ${ratio('#FFFFFF','#0A8F5F')}:1 | glyph #34C68C on #0B1116: ${ratio('#34C68C','#0B1116')}:1`)
})()
